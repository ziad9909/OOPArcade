35char Ultimate_TTT_Board::check_sub_board_win(int main_r, int main_c) {
36 const auto& sub = sub_boards[main_r][main_c];
37
38 // Helper lambda to check if three cells are equal and not blank
39 auto all_equal = [&](int r1, int c1, int r2, int c2, int r3, int c3) {
40 return sub[r1][c1] == sub[r2][c2] &&
41 sub[r2][c2] == sub[r3][c3] &&
42 sub[r1][c1] != blank_symbol;
43 };
44
45 // Check rows
46 for (int i = 0; i < 3; i++) {
47 if (all_equal(i, 0, i, 1, i, 2)) {
48 return sub[i][0];
49 }
50 }
51
52 // Check columns
53 for (int j = 0; j < 3; j++) {
54 if (all_equal(0, j, 1, j, 2, j)) {
55 return sub[0][j];
56 }
57 }
58
59 // Check diagonals
60 if (all_equal(0, 0, 1, 1, 2, 2)) {
61 return sub[1][1];
62 }
63 if (all_equal(0, 2, 1, 1, 2, 0)) {
64 return sub[1][1];
65 }
66
67 return blank_symbol; // No winner
68}

96bool Ultimate_TTT_Board::update_board(Move<char>* move) {
97 int global_x = move->get_x();
98 int global_y = move->get_y();
99 char mark = move->get_symbol();
100
101 // Validate global coordinates
102 if (global_x < 0 || global_x >= 9 || global_y < 0 || global_y >= 9) {
103 return false;
104 }
105
106 // Convert to main and sub coordinates
107 int main_r, main_c, sub_r, sub_c;
108 tie(main_r, main_c, sub_r, sub_c) = convert_coordinates(global_x, global_y);
109
110 // Check if this sub-board is the active one (or if any sub-board is allowed)
111 if (active_main_row != -1 && active_main_col != -1) {
112 if (main_r != active_main_row || main_c != active_main_col) {
113 cout << "Invalid move! You must play in sub-board ("
114 << active_main_row << ", " << active_main_col << ")\n";
115 return false;
116 }
117 }
118
119 // Check if sub-board is available
120 if (!is_sub_board_available(main_r, main_c)) {
121 cout << "Invalid move! Sub-board (" << main_r << ", " << main_c
122 << ") is already won or full.\n";
123 return false;
124 }
125
126 // Check if cell in sub-board is empty
127 if (sub_boards[main_r][main_c][sub_r][sub_c] != blank_symbol) {
128 cout << "Invalid move! Cell already occupied.\n";
129 return false;
130 }
131
132 // Make the move
133 sub_boards[main_r][main_c][sub_r][sub_c] = toupper(mark);
134 sub_board_moves[main_r][main_c]++;
135 n_moves++;
136
137 // Update main board representation (for compatibility)
138 board[global_x][global_y] = toupper(mark);
139
140 // Check if this sub-board has a winner
141 char winner = check_sub_board_win(main_r, main_c);
142 if (winner != blank_symbol) {
143 main_board[main_r][main_c] = winner;
144 cout << "\n*** Sub-board (" << main_r << ", " << main_c
145 << ") won by " << winner << "! ***\n";
146 } else if (is_sub_board_full(main_r, main_c)) {
147 // Sub-board is full but no winner (draw)
148 main_board[main_r][main_c] = 'D'; // Mark as drawn
149 cout << "\n*** Sub-board (" << main_r << ", " << main_c
150 << ") ended in a draw ***\n";
151 }
152
153 // Determine next active sub-board
154 // If the target sub-board (determined by sub_r, sub_c) is available, use it
155 // Otherwise, player can choose any available sub-board
156 if (is_sub_board_available(sub_r, sub_c)) {
157 active_main_row = sub_r;
158 active_main_col = sub_c;
159 } else {
160 // Target sub-board is won or full, player can choose any available
161 active_main_row = -1;
162 active_main_col = -1;
163 }
164
165 return true;
166}

168bool Ultimate_TTT_Board::is_win(Player<char>* player) {
169 const char sym = player->get_symbol();
170
171 // Check if player has won three sub-boards in a row on main board
172 auto all_equal = [&](char a, char b, char c) {
173 return a == b && b == c && a == sym;
174 };
175
176 // Check rows
177 for (int i = 0; i < 3; i++) {
178 if (all_equal(main_board[i][0], main_board[i][1], main_board[i][2])) {
179 return true;
180 }
181 }
182
183 // Check columns
184 for (int j = 0; j < 3; j++) {
185 if (all_equal(main_board[0][j], main_board[1][j], main_board[2][j])) {
186 return true;
187 }
188 }
189
190 // Check diagonals
191 if (all_equal(main_board[0][0], main_board[1][1], main_board[2][2])) {
192 return true;
193 }
194 if (all_equal(main_board[0][2], main_board[1][1], main_board[2][0])) {
195 return true;
196 }
197
198 return false;
199}

201bool Ultimate_TTT_Board::is_draw(Player<char>* player) {
202 // Check if all sub-boards are won/drawn and no main board winner
203 bool all_claimed = true;
204 for (int i = 0; i < 3; i++) {
205 for (int j = 0; j < 3; j++) {
206 if (main_board[i][j] == blank_symbol) {
207 all_claimed = false;
208 break;
209 }
210 }
211 if (!all_claimed) break;
212 }
213
214 return all_claimed && !is_win(player);
215}
