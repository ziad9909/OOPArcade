140 * @brief Check if player has won
141 * @param player Pointer to player
142 * @return true if player satisfies diamond TTT winning condition
143 */
144bool Diamond_TTT_Board::is_win(Player<char>* player) {
145 char symbol = player->get_symbol();
146 auto all_lines = get_all_lines();
147 vector<vector<pair<int, int>>> completed_3lines;
148 vector<vector<pair<int, int>>> completed_4lines;
149
150 // Collect all completed lines for this player
151 for (const auto& line : all_lines) {
152 bool complete = true;
153 for (const auto& cell : line)
154 if (board[cell.first][cell.second] != symbol) {
155 complete = false;
156 break;
157 }
158
159 if (complete) {
160 if (line.size() == 3) completed_3lines.push_back(line);
161 else if (line.size() == 4) completed_4lines.push_back(line);
162 }
163 }
164
165 // Need at least one 3-line and one 4-line
166 if (completed_3lines.empty() || completed_4lines.empty()) return false;
167
168 // Check all combinations of 3-lines and 4-lines
169 for (const auto& three_line : completed_3lines) {
170 // Determine direction of the 3-line
171 string dir3 = "other";
172 if (three_line.size() > 1) {
173 int dx = three_line[1].first - three_line[0].first;
174 int dy = three_line[1].second - three_line[0].second;
175 if (dx == 0) dir3 = "horizontal";
176 else if (dy == 0) dir3 = "vertical";
177 else if (abs(dx) == abs(dy)) dir3 = "diagonal";
178 }
179
180 for (const auto& four_line : completed_4lines) {
181 // Determine direction of the 4-line
182 string dir4 = "other";
183 if (four_line.size() > 1) {
184 int dx = four_line[1].first - four_line[0].first;
185 int dy = four_line[1].second - four_line[0].second;
186 if (dx == 0) dir4 = "horizontal";
187 else if (dy == 0) dir4 = "vertical";
188 else if (abs(dx) == abs(dy)) dir4 = "diagonal";
189 }
190
191 // Must be in different directions
192 if (dir3 == dir4) continue;
193
194 // Check if they share any cell
195 bool shares_cell = false;
196 for (const auto& cell3 : three_line) {
197 for (const auto& cell4 : four_line) {
198 if (cell3.first == cell4.first && cell3.second == cell4.second) {
199 shares_cell = true;
200 break;
201 }
202 }
203 if (shares_cell) break;
204 }
205
206 // They must NOT share any cell
207 if (!shares_cell) {
208 return true;
209 }
210 }
211 }
212 return false;
213}

358int Diamond_TTT_UI::minimax_helper(Diamond_TTT_Board* board,
359 char current_symbol,
360 char ai_symbol,
361 char opponent_symbol,
362 bool isMaximizing,
363 int depth)
364{
365 int score = evaluate_board(board, ai_symbol, opponent_symbol);
366
367 if (score == 10 || score == -10)
368 return score;
369
370 if (board->get_empty_cells().empty())
371 return 0;
372
373 if (depth == 0)
374 return score;
375
376 if (isMaximizing) {
377 int bestVal = INT_MIN;
378
379 auto empty_cells = board->get_empty_cells();
380 for (auto cell : empty_cells) {
381 board->update_board(new Move<char>(cell.first, cell.second, current_symbol));
382
383 int value = minimax_helper(board,
384 opponent_symbol,
385 ai_symbol,
386 opponent_symbol,
387 false,
388 depth - 1);
389
390 board->undo_move(cell.first, cell.second);
391 bestVal = max(bestVal, value);
392 }
393 return bestVal;
394 }
395 else {
396 int bestVal = INT_MAX;
397
398 auto empty_cells = board->get_empty_cells();
399 for (auto cell : empty_cells) {
400 board->update_board(new Move<char>(cell.first, cell.second, current_symbol));
401
402 int value = minimax_helper(board,
403 ai_symbol,
404 ai_symbol,
405 opponent_symbol,
406 true,
407 depth - 1);
408
409 board->undo_move(cell.first, cell.second);
410 bestVal = min(bestVal, value);
411 }
412 return bestVal;

447 * @brief Find best move for AI
448 */
449pair<int, int> Diamond_TTT_UI::find_best_move(Diamond_TTT_Board* board, char ai_symbol, char opponent_symbol) {
450 int bestVal = INT_MIN;
451 pair<int, int> bestMove = { -1, -1 };
452
453 auto empty_cells = board->get_empty_cells();
454
455 // If there are too many empty cells, use a heuristic
456 if (empty_cells.size() > 10) {
457 // Simple heuristic: prefer center cells first
458 vector<pair<int, int>> center_cells = { {0, 2}, {1, 1}, {1, 2}, {1, 3}, {2, 0}, {2, 1}, {2, 2}, {2, 3}, {2, 4}, {3, 1}, {3, 2}, {3, 3}, {4, 2} };
459
460 // Check for winning move
461 for (auto cell : empty_cells) {
462 board->update_board(new Move<char>(cell.first, cell.second, ai_symbol));
463 Player<char> temp_ai("AI", ai_symbol, PlayerType::COMPUTER);
464 if (board->is_win(&temp_ai)) {
465 board->undo_move(cell.first, cell.second);
466 return cell;
467 }
468 board->undo_move(cell.first, cell.second);
469 }
470
471 // Check for blocking opponent's winning move
472 for (auto cell : empty_cells) {
473 board->update_board(new Move<char>(cell.first, cell.second, opponent_symbol));
474 Player<char> temp_opp("OPP", opponent_symbol, PlayerType::COMPUTER);
475 if (board->is_win(&temp_opp)) {
476 board->undo_move(cell.first, cell.second);
477 return cell;
478 }
479 board->undo_move(cell.first, cell.second);
480 }
481
482 // Prefer center cells
483 for (auto cell : center_cells) {
484 if (board->validate_move(cell.first, cell.second).empty()) {
485 return cell;
486 }
487 }
488
489 // Otherwise, return first available cell
490 if (!empty_cells.empty()) {
491 return empty_cells[0];
492 }
493 }
494
495 // For fewer empty cells, use minimax
496 for (auto cell : empty_cells) {
497 board->update_board(new Move<char>(cell.first, cell.second, ai_symbol));
498 int moveVal = minimax(board, opponent_symbol, ai_symbol, opponent_symbol, false);
499 board->undo_move(cell.first, cell.second);
500
501 if (moveVal > bestVal) {
502 bestVal = moveVal;
503 bestMove = cell;
504 }
505
506 // If we find a winning move, take it immediately
507 if (moveVal == 10) {
508 break;
509 }
510 }
511
512 // If no good move found, pick first available
513 if (bestMove.first == -1 && !empty_cells.empty()) {
514 bestMove = empty_cells[0];
515 }
516
517 return bestMove;
518}
