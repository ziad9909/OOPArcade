159 * @param symbol Computer player's symbol ('X' or 'O')

167 /**
168 * @brief Generate a random valid move for the computer player
169 * @return Pointer to Move object with random valid coordinates
170 *
171 * Randomly searches for empty cells ('_') on the board
172 * and returns a move to place the computer's symbol there.
173 * Continues searching until a valid empty cell is found.
174 */
175 Move<char>* get_computer_move() {
176 Board<char>* board = this->get_board_ptr();
177 while (true) {
178 int x = rand() % board->get_rows();
179 int y = rand() % board->get_columns();
180 if (board->get_cell(x, y) == '_') {
181 return new Move<char>(x, y, this->symbol);
182 }
183 }
184 }
