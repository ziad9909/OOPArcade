95int Four_in_a_row_board::count_sequences(char symbol, int length) {
96 int count = 0;
97
98 // Helper lambda to check sequence
99 auto check_sequence = [&](int r, int c, int dr, int dc) -> bool {
100 for (int i = 0; i < length; i++) {
101 int nr = r + i * dr;
102 int nc = c + i * dc;
103 if (nr < 0 || nr >= rows || nc < 0 || nc >= columns) return false;
104 if (board[nr][nc] != symbol) return false;
105 }
106 return true;
107 };
108
109 // Check all positions and directions
110 for (int r = 0; r < rows; r++) {
111 for (int c = 0; c < columns; c++) {
112 // Horizontal
113 if (check_sequence(r, c, 0, 1)) count++;
114 // Vertical
115 if (check_sequence(r, c, 1, 0)) count++;
116 // Diagonal (backslash)
117 if (check_sequence(r, c, 1, 1)) count++;
118 // Diagonal (forward slash)
119 if (check_sequence(r, c, -1, 1)) count++;
120 }
121 }
122
123 return count;
